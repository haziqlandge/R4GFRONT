# Runtime dependencies for services/. Python 3.12.
# Distinct from requirements-dev.txt, which carries the offline build and
# measurement tooling. Keep this list minimal: every import here is loaded
# inside the 200ms budget's process.

# Inference. Rules.md 2.1: ONNX Runtime only, never PyTorch at request time.
onnxruntime>=1.19

# Tokenization. The `tokenizers` Rust library loads tokenizer.json directly,
# so neither `transformers` nor `torch` is needed. Smaller image, faster start.
tokenizers>=0.20

# Indexes
hnswlib>=0.8
numpy>=1.26

# Service
fastapi>=0.115
uvicorn[standard]>=0.30

# WebSocket CLIENT, for the stt_gateway relay to Sarvam's realtime socket
# (/v1/stt/live). uvicorn[standard] already installs this as a websocket SERVER
# implementation, so on a machine that has the line above it is present either
# way - but the gateway imports it directly now, and an import that works by
# accident is one dependency resolution away from not working.
websockets>=13

# Required by stt_gateway's POST /v1/stt/file, which takes an audio upload.
# FastAPI resolves Form/UploadFile parameters when it REGISTERS the route, not
# when the route is called, so without this the whole service fails to start -
# not just that one endpoint. It is a hard runtime dependency despite never
# being imported by our own code, which is why it is easy to omit.
python-multipart>=0.0.9
pydantic>=2.8
orjson>=3.10
httpx>=0.27

# Artifact loading at startup
